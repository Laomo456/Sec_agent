### Step 3 — 写 Hook 脚本

在本地（如 `C:\Users\20751\Desktop\快手test\`）创建 `live_capture.js`：

```javascript
// live_capture.js — 快手网络抓包脚本
(function() {
    // ========== 工具函数 ==========
    function ts() {
        var d = new Date();
        return ("0"+d.getHours()).slice(-2)+":"+("0"+d.getMinutes()).slice(-2)+":"+
               ("0"+d.getSeconds()).slice(-2)+"."+("00"+d.getMilliseconds()).slice(-3);
    }

    // ========== 第1层: Java OkHttp Hook ==========
    Java.perform(function() {

        // 拦截 OkHttp 请求
        var OkHttpClient = Java.use("okhttp3.OkHttpClient");
        OkHttpClient.newCall.overload("okhttp3.Request").implementation = function(request) {
            var url = request.url().toString();
            var method = request.method();
            console.log("\n>>> [" + ts() + "] REQ " + method + " " + url);
            return this.newCall(request);
        };

        // 拦截 OkHttp 响应
        var RealCall = Java.use("okhttp3.RealCall");
        RealCall.getResponseWithInterceptorChain.overload().implementation = function() {
            var resp = this.getResponseWithInterceptorChain();
            var code = resp.code();
            var req = resp.request();
            var body = resp.body();
            var len = body ? body.contentLength() : -1;
            console.log("<<< [" + ts() + "] RESP " + code + " " + req.url() + " (" + len + "B)");
            return resp;
        };
    });

    // ========== 第2层: Native TCP connect Hook ==========
    var libc = Process.findModuleByName("libc.so");
    libc.enumerateExports().forEach(function(e) {
        if (e.name === "connect") {
            Interceptor.attach(e.address, {
                onEnter: function(args) {
                    var fd = args[0].toInt32();
                    var addr = args[1];
                    var family = addr.readU16();
                    if (family === 2) { // AF_INET
                        var port = (addr.readU8(2) << 8) + addr.readU8(3);
                        var ip = addr.readU8(4) + "." + addr.readU8(5) + "." +
                                 addr.readU8(6) + "." + addr.readU8(7);
                        console.log("[TCP] [" + ts() + "] connect " + ip + ":" + port + " fd=" + fd);
                    }
                }
            });
        }
    });

    // ========== 第3层: Native send/recv Hook ==========
    // send
    libc.enumerateExports().forEach(function(e) {
        if (e.name === "send") {
            Interceptor.attach(e.address, {
                onEnter: function(args) {
                    var fd = args[0].toInt32();
                    var buf = args[1];
                    var len = args[2].toInt32();
                    if (len > 0 && len < 8192 && fd > 50) {
                        var data = buf.readByteArray(len);
                        // 尝试当文本打印
                        try {
                            var text = "";
                            for (var j = 0; j < Math.min(len, 256); j++)
                                text += String.fromCharCode(data[j] & 0xFF);
                            if (text.startsWith("GET ") || text.startsWith("POST") || text.startsWith("HTTP")) {
                                console.log("[SEND:"+fd+"] [" + ts() + "]\n" + text);
                            } else {
                                var hex = "";
                                for (var j = 0; j < Math.min(len, 64); j++)
                                    hex += ('0' + (data[j] & 0xFF).toString(16)).slice(-2);
                                console.log("[SEND:"+fd+"] [" + ts() + "] " + len + "B hex=" + hex);
                            }
                        } catch(e) {}
                    }
                }
            });
        }
    });
    // recv
    libc.enumerateExports().forEach(function(e) {
        if (e.name === "recv") {
            Interceptor.attach(e.address, {
                onEnter: function(args) {
                    this.fd = args[0].toInt32();
                    this.buf = args[1];
                    this.len = args[2].toInt32();
                },
                onLeave: function(retval) {
                    var n = retval.toInt32();
                    if (n > 0 && n < 8192 && this.fd > 50) {
                        var data = this.buf.readByteArray(n);
                        try {
                            var text = "";
                            for (var j = 0; j < Math.min(n, 256); j++)
                                text += String.fromCharCode(data[j] & 0xFF);
                            if (text.startsWith("HTTP")) {
                                console.log("[RECV:"+this.fd+"] [" + ts() + "]\n" + text);
                            } else {
                                var hex = "";
                                for (var j = 0; j < Math.min(n, 64); j++)
                                    hex += ('0' + (data[j] & 0xFF).toString(16)).slice(-2);
                                console.log("[RECV:"+this.fd+"] [" + ts() + "] " + n + "B hex=" + hex);
                            }
                        } catch(e) {}
                    }
                }
            });
        }
    });

    console.log("[+] live_capture 全部 Hook 就绪");
})();
```

## 二、三层 Hook 架构说明

```
┌─────────────────────────────────────────────────┐
│  live_capture.js                                 │
│                                                  │
│  第1层 Java Hook (Java.perform)                  │
│  ┌─────────────────────────────────────────────┐ │
│  │ OkHttpClient.newCall()        → 打印 REQ    │ │
│  │ RealCall.getResponseWith...() → 打印 RESP   │ │
│  └─────────────────────────────────────────────┘ │
│             ↓ 捕获 OkHttp 发出的 HTTP 请求       │
│                                                  │
│  第2层 Native Hook (libc.so connect)             │
│  ┌─────────────────────────────────────────────┐ │
│  │ connect(AF_INET)              → 打印 ip:port│ │
│  └─────────────────────────────────────────────┘ │
│             ↓ 捕获所有 TCP 出站连接              │
│                                                  │
│  第3层 Native Hook (libc.so send/recv)           │
│  ┌─────────────────────────────────────────────┐ │
│  │ send()  HTTP→打印明文  二进制→hex           │ │
│  │ recv()  同上                                 │ │
│  └─────────────────────────────────────────────┘ │
│             ↓ 捕获 socket 上实际收发数据         │
└─────────────────────────────────────────────────┘
```

### 三层的作用

| 层 | 捕获目标 | 比如能抓到 |
|----|---------|-----------|
| 1 - Java OkHttp | HTTP 请求/响应 header + URL | `POST https://az3-api.ksapisrv.com/rest/n/feed/photo/info?sig=...` |
| 2 - connect | 所有 TCP 连接的目标地址 | `connect 2.2.2.2:514`、`connect 47.93.x.x:443` |
| 3 - send/recv | socket 上传输的原始字节 | HTTP 请求体、响应体、Klink 二进制包 |

### 3.2 URL 替换机制（重要发现）

```
>>> REQ  OkHttp层面看到: https://api.mock-host.com/rest/n/feed/photo/info
<<< RESP 实际走了:      https://az3-api.ksapisrv.com/rest/n/feed/photo/info?...
```

`com.kuaishou.aegon.okhttp.a` 在 OkHttp 拦截器链中把域名替换了。
这说明 Aegon 在起作用，不是死代码。

## 四、⚠️ 遗留问题

### 4.1 2.2.2.2:514 海量连接
- **现象**：每秒 4-10 次 `connect 2.2.2.2:514`，持续不断
- **特征**：同一组 fd 循环重连，4-5 个 socket 并发
- **原因未知**：当前 send/recv hook 看不到这些 socket 上跑的协议
- **猜测**：Klink 心跳 / SDK 探测 / 独立日志通道

### 4.2 未 Hook 的点
- **Cronet/QUIC** — `CronetInterceptor` 存在，可能视频流量走 QUIC
- **body 明文** — 当前 RealCall hook 没 dump response body.string()
- **ARM SO 延迟加载** — libklink/libksse/libaegon 在 x86_64 模拟器上可能走 NativeBridge 晚加载

### 4.3 只测了滑动 feed
点进视频、搜索、个人主页、IM、直播这些操作会产生更多 API，还没触发。
